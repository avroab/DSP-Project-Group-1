           end

        % Button pushed function: EvaluateButton
        function EvaluateButtonPushed(app, event)
            if app.flag_single_omr==1
                app.ResultPanel.Visible=true;
                app.InsertedOMRSheetPanel.Visible=true;
                app.RequiredInformationsPanel.Visible=false;
                app.ResultDetailsPanel.Visible=false;
                app.ResultSummaryPanel.Visible=false;
                app.InsertOMROptionsPanel.Visible=false;
                app.ResultDetailsPanel.Visible=false;
                [roll,roll_valid]=get_roll(app.single_p_img);
                app.Roll_resultPanel.Value=roll; 
                  %  getting roll from get_roll function.
                app.Score_resultPanel.Value=evaluate_single_omr(app.single_p_img,app.soln,app.n_question);
                    % geting score from evaluate_single_omr function
                [code,code_valid]=get_code(app.single_p_img);
                app.SubCode_resultPanel.Value=code;
                    %  getting code from get_code function.
                score= app.Score_resultPanel.Value;

                if score < app.Score2Pass.Value
                    app.Remarks.Value="Fail!";
                elseif score >= app.Score2Pass.Value
                    app.Remarks.Value="Pass!";
                end
                
                if (code_valid==0)
                    app.Remarks.Value="code is invalid";
                elseif (roll_valid==0)
                    app.Remarks.Value="Roll is invalid";
                elseif app.SubCode.Value~=app.SubCode_resultPanel.Value
                    app.Remarks.Value="SubCode Mismatched";
                end
                app.messageArea.Value="The inserted OMR sheet is evaluated Successfully!";
            elseif app.flag_multiple_omr==1
                f=app.multiple_omr_dir;
                [roll_array,code_array,marks_array,reject]=evaluate_multiple_omr ...
                    (f,app.soln,app.n_question,app.SubCode.Value);
                app.ResultDetailsPanel.Visible=true;
                app.ResultPanel.Visible=false;
                app.InsertedOMRSheetPanel.Visible=false;
                app.RequiredInformationsPanel.Visible=false;
                app.InsertOMROptionsPanel.Visible=false;
                roll_str=num2str(roll_array);
                [row,col]=size(roll_str);
                high=0;
                total_score=0;
                passed=0;
                for del_ind=1:10
                app.UITable.Data{del_ind,1}=[];
                app.UITable.Data{del_ind,2}=[];
                end
                 for index=1:row % indexing all the evaluated omr informations
                    app.UITable.Data{index,1} = roll_str(index,:); 
                            %inerting roll string in the 1st row of the table 
                    app.UITable.Data{index,2} = marks_array(index);
                            %inerting score in the 2nd row of the table 
                    total_score=total_score+marks_array(index);
                    if marks_array(index)>high
                        high=marks_array(index);
                    end
                    if marks_array(index)>=app.Score2Pass.Value
                        passed=passed+1;
                    end
                 end
                if row~=0
                xlswrite('result.xls',cell(30,1),'Sheet1','A2:A100');
                xlswrite('result.xls',cell(30,1),'Sheet1','B2:B100');
                xlswrite('result.xls',cell(30,1),'Sheet1','C2:C100');
                xlswrite('result.xls',roll_array,'Sheet1','A2');
                xlswrite('result.xls',code_array,'Sheet1','B2');
                xlswrite('result.xls',marks_array,'Sheet1','C2');
                end
               
                app.ResultSummaryPanel.Visible=true;
                app.TotalEvaluated.Value=row;
                app.TotalReject.Value=reject;
                if row~=0
                    app.AverageScore.Value=total_score/row;
                else 
                    app.AverageScore.Value=0;
                end
                app.HighestScore.Value=high;
                app.PassedStudent.Value=passed;
                app.messageArea.Value="The inserted OMR sheets are evaluated Successfully!";
            end
        end

        % Construct app
        function app = OMR_Sheet_Evaluator

            % Create UIFigure and components
            createComponents(app)

            % Register the app with App Designer
            registerApp(app, app.OMRSheetEvaluatorUIFigure)

            % Execute the startup function
            runStartupFcn(app, @startupFcn)

            if nargout == 0
                clear app
            end
        end

        function delete(app)

            % Delete UIFigure when app is deleted
            delete(app.OMRSheetEvaluatorUIFigure)
        end
